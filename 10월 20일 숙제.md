# 햄버거 분배
어떻게 이해하고 접근했는지 이따가 알려드려도 될까요?
```
N, K = map(int, input().split())
PH = list(input())
count = 0

for i in range(N):
    if PH[i] == "P":
        start = max(0, i - K)
        end = min(N, i + K + 1)
        
        for j in range(start, end):
            if PH[j] == "H":
                PH[j] = "N" 
                count += 1
                break       

print(count)
```
# 냉장고
```
import sys

input = sys.stdin.read

def solve():
    data = input().split()
    if not data:
        return
    
    N = int(data[0])
    substances = []
    idx = 1
    for _ in range(N):
        x = int(data[idx])
        y = int(data[idx+1])
        substances.append((x, y))
        idx += 2
        
    substances.sort(key=lambda item: item[1])
    
    count = 1
```
