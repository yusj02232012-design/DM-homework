```
import sys
input = sys.stdin.readline

N = int(input())
A = list(map(int, input().split()))

def solve(N,A):
    bdp = 0 if A[0] > A[N-1] else N-1
    if N <= 3:
        bdp = A.index(max(A))
    b1p = 0 if bdp != 0 else N-1
    max_ = 0
    for  i in range(1, N-1):
        su = sum(A) -A[b1p]-A[i]
        for j in range(i,bdp+1):
            if i == j:
                continue
            su += A[j]
        if su > max_:
            max_ = su

    return max_
        
print(solve(N,A))
```
